let scores = { chaotic: 0, focused: 0, social: 0, efficient: 0 };

let current = 0;


function getResult() {

  const sorted = Object.entries(scores).sort((a, b) => b[1] - a[1]);

  const top = sorted[0][0], second = sorted[1][0];

  return results.find(r => { const k = Object.keys(r.match); return k.length === 2 && k.includes(top) && k.includes(second); })

    || results.find(r => Object.keys(r.match).length === 1 && Object.keys(r.match)[0] === top);

}


function renderQuiz() {

  const q = questions[current];

  const pips = questions.map((_, i) => `<div class="pip ${i <= current ? 'done' : ''}"></div>`).join('');

  document.getElementById('app').innerHTML = `

    <div class="label">What AI Model Are You?</div>

    <div class="progress">${pips}</div>

    <h2>${q.q}</h2>

    <div class="options">${q.options.map((opt, i) => `<button class="opt" onclick="answer(${i})">${opt.text}</button>`).join('')}</div>

    <div class="counter">Question ${current + 1} of ${questions.length}</div>

  `;

  window._opts = q.options;

}


function answer(i) {

  document.querySelectorAll('.opt')[i].classList.add('selected');

  setTimeout(() => {

    Object.entries(window._opts[i].traits).forEach(([t, v]) => scores[t] += v);

    current++;

    if (current >= questions.length) renderResult();

    else renderQuiz();

  }, 300);

}


function renderResult() {

  const r = getResult();

  document.getElementById('app').innerHTML = `

    <div class="result">

      <div class="emoji">${r.emoji}</div>

      <div class="label">You are...</div>

      <h1>${r.model}</h1>

      <p class="tagline">"${r.tagline}"</p>

      <p class="desc">${r.desc}</p>

      <a class="hf" href="${r.url}" target="_blank">View on HuggingFace</a><br>

      <button class="retry" onclick="reset()">Retake Quiz</button>

    </div>

  `;

}


function reset() {

  scores = { chaotic: 0, focused: 0, social: 0, efficient: 0 };

  current = 0;

  renderQuiz();

}
